 * 3. Bam Save.
 * 4. Quay lai Sheet, dung nhu ham co san:
 *      =REG60(A1)

 *
 * LUU Y VE ENCODING:

function md5_safeAdd(x, y) {
  var lsw = (x & 0xffff) + (y & 0xffff);
  var msw = (x >> 16) + (y >> 16) + (lsw >> 16);
  return (msw << 16) | (lsw & 0xffff);
}
function md5_rotl(num, cnt) {
  return (num << cnt) | (num >>> (32 - cnt));
}
function md5_cmn(q, a, b, x, s, t) {
  return md5_safeAdd(md5_rotl(md5_safeAdd(md5_safeAdd(a, q), md5_safeAdd(x, t)), s), b);
}
function md5_ff(a, b, c, d, x, s, t) { return md5_cmn((b & c) | (~b & d), a, b, x, s, t); }
function md5_gg(a, b, c, d, x, s, t) { return md5_cmn((b & d) | (c & ~d), a, b, x, s, t); }
function md5_hh(a, b, c, d, x, s, t) { return md5_cmn(b ^ c ^ d, a, b, x, s, t); }
function md5_ii(a, b, c, d, x, s, t) { return md5_cmn(c ^ (b | ~d), a, b, x, s, t); }

function md5_binlToRstr(input) {
  var i, output = '';
  var length32 = input.length * 32;
  for (i = 0; i < length32; i += 8) {
    output += String.fromCharCode((input[i >> 5] >>> (i % 32)) & 0xff);
  }
  return output;
}
function md5_rstrToBinl(input) {
  var i, output = [];
  output[(input.length >> 2) - 1] = undefined;
  for (i = 0; i < output.length; i += 1) output[i] = 0;
  var length8 = input.length * 8;
  for (i = 0; i < length8; i += 8) {
    output[i >> 5] |= (input.charCodeAt(i / 8) & 0xff) << (i % 32);
  }
  return output;
}
function md5_rstr(s) {
  return md5_binlToRstr(md5_binl(md5_rstrToBinl(s), s.length * 8));
}
function md5_rstrToHex(input) {
  var hexTab = '0123456789abcdef';
  var output = '';
  var x;
  for (var i = 0; i < input.length; i += 1) {
    x = input.charCodeAt(i);
    output += hexTab.charAt((x >>> 4) & 0x0f) + hexTab.charAt(x & 0x0f);
  }
  return output;
}
function md5_toRawBytes(str) {
  // Chuoi ASCII/khong dau -> 1 ky tu = 1 byte, giong StrConv(...,vbFromUnicode)
  // voi chuoi ASCII thuan tuy.
  var out = '';
  for (var i = 0; i < str.length; i++) {
    out += String.fromCharCode(str.charCodeAt(i) & 0xff);
  }
  return out;
}
function MD5_HEX(str) {
  return md5_rstrToHex(md5_rstr(md5_toRawBytes(String(str))));
}

function repeat8(arr) { return arr.concat(arr, arr, arr); }
var HAVAL_M0 = repeat8([0, 7, 6, 5, 4, 3, 2, 1]);
var HAVAL_M1 = repeat8([1, 0, 7, 6, 5, 4, 3, 2]);
var HAVAL_M2 = repeat8([2, 1, 0, 7, 6, 5, 4, 3]);

function rotr32(x, n) {
  x = x >>> 0;
  return ((x >>> n) | (x << (32 - n))) >>> 0;
}
function haval_f1(x6, x5, x4, x3, x2, x1, x0) {
  return ((x1 & x4) ^ (x2 & x5) ^ (x3 & x6) ^ (x0 & x1) ^ x0) >>> 0;
}
function haval_f2(x6, x5, x4, x3, x2, x1, x0) {
  return (((x1 & x2 & x3) ^ (x2 & x4 & x5) ^ (x1 & x2) ^ (x1 & x4) ^
    (x2 & x6) ^ (x3 & x5) ^ (x4 & x5) ^ (x0 & x2) ^ x0)) >>> 0;
}
function haval_f3(x6, x5, x4, x3, x2, x1, x0) {
  return ((x1 & x2 & x3) ^ (x1 & x4) ^ (x2 & x5) ^ (x3 & x6) ^ (x0 & x3) ^ x0) >>> 0;
}
function bytesToWordsLE(block) {
  var x = new Array(32);
  for (var i = 0; i < 32; i++) {
    var j = i * 4;
    x[i] = (block[j] | (block[j + 1] << 8) | (block[j + 2] << 16) | (block[j + 3] << 24)) >>> 0;
  }
  return x;
}
function havalTransform3(state, block) {
  var x = bytesToWordsLE(block);
  var E = state.slice();
  var i, val;
  for (i = 0; i < 32; i++) {
    val = (rotr32(haval_f1(E[HAVAL_M1[i]], E[HAVAL_M0[i]], E[HAVAL_M3[i]], E[HAVAL_M5[i]], E[HAVAL_M6[i]], E[HAVAL_M2[i]], E[HAVAL_M4[i]]), 7)
      + rotr32(E[HAVAL_M7[i]], 11) + x[i]) >>> 0;
    E[7 - (i % 8)] = val;
  }
  for (i = 0; i < 32; i++) {
    val = (rotr32(haval_f2(E[HAVAL_M4[i]], E[HAVAL_M2[i]], E[HAVAL_M1[i]], E[HAVAL_M0[i]], E[HAVAL_M5[i]], E[HAVAL_M3[i]], E[HAVAL_M6[i]]), 7)
      + rotr32(E[HAVAL_M7[i]], 11) + x[HAVAL_I2[i]] + HAVAL_K2[i]) >>> 0;
    E[7 - (i % 8)] = val;
  }
  for (i = 0; i < 32; i++) {
    val = (rotr32(haval_f3(E[HAVAL_M6[i]], E[HAVAL_M1[i]], E[HAVAL_M2[i]], E[HAVAL_M3[i]], E[HAVAL_M4[i]], E[HAVAL_M5[i]], E[HAVAL_M0[i]]), 7)
      + rotr32(E[HAVAL_M7[i]], 11) + x[HAVAL_I3[i]] + HAVAL_K3[i]) >>> 0;
    E[7 - (i % 8)] = val;
  }
  for (i = 0; i < 8; i++) {
    state[i] = (state[i] + E[i]) >>> 0;
  }
}
function stringToAsciiBytes(str) {
  var bytes = [];
  for (var i = 0; i < str.length; i++) {
    bytes.push(str.charCodeAt(i) & 0xff);
  }
  return bytes;
}
function haval128_3Hex(messageBytes) {
  var state = HAVAL_D0.slice();
  var msgLenBits = messageBytes.length * 8;
  var data = messageBytes.slice();
  data.push(0x01);
  while (data.length % 128 !== 118) {
    data.push(0x00);
  }
  var b0 = (1 & 0x07) | ((3 & 0x07) << 3) | ((128 & 0x03) << 6);
  var b1 = (128 >> 2) & 0xFF;
  var count0 = msgLenBits >>> 0;
  var count1 = Math.floor(msgLenBits / 4294967296) >>> 0;
  data.push(b0, b1);
  data.push(count0 & 0xff, (count0 >>> 8) & 0xff, (count0 >>> 16) & 0xff, (count0 >>> 24) & 0xff);
  data.push(count1 & 0xff, (count1 >>> 8) & 0xff, (count1 >>> 16) & 0xff, (count1 >>> 24) & 0xff);
  for (var i = 0; i < data.length; i += 128) {
    havalTransform3(state, data.slice(i, i + 128));
  }
  var s0 = state[0], s1 = state[1], s2 = state[2], s3 = state[3];
  var s4 = state[4], s5 = state[5], s6 = state[6], s7 = state[7];
  s3 = (s3 + ((s7 & 0xFF000000) | (s6 & 0x00FF0000) | (s5 & 0x0000FF00) | (s4 & 0x000000FF))) >>> 0;
  s2 = (s2 + ((((s7 & 0x00FF0000) | (s6 & 0x0000FF00) | (s5 & 0x000000FF)) << 8) |
    ((s4 & 0xFF000000) >>> 24))) >>> 0;
  s1 = (s1 + ((((s7 & 0x0000FF00) | (s6 & 0x000000FF)) << 16) |
    (((s5 & 0xFF000000) | (s4 & 0x00FF0000)) >>> 16))) >>> 0;
  s0 = (s0 + (((s7 & 0x000000FF) << 24) |
    (((s6 & 0xFF000000) | (s5 & 0x00FF0000) | (s4 & 0x0000FF00)) >>> 8))) >>> 0;
  function toHexLE(w) {
    var out = '';
    for (var k = 0; k < 4; k++) {
      var b = (w >>> (8 * k)) & 0xff;
      out += ('0' + b.toString(16)).slice(-2);
    }
    return out;
  }
  return toHexLE(s0) + toHexLE(s1) + toHexLE(s2) + toHexLE(s3);
}
function reg60Core(message) {
  var parts = String(message).split('-');
  var transformed = parts[0] + '-' + (parseInt(parts[1], 10) + 10) + '-' + parts[2];
  return haval128_3Hex(stringToAsciiBytes(transformed));
}

/**
 * Tuong duong reg60() trong VBA cu (dung HAVAL-128/3, co +10 vao phan giua).
 * Vi du: =REG60(A1)  voi A1 = "591523-2147154943-1075053569"
 * @param {string} input Chuoi dang "so1-so2-so3"
 * @return {string} Ma HAVAL-128/3 dang hex chu HOA
 * @customfunction
 */
function REG60(input) {
  if (input === undefined || input === null || input === '') return '';
  return reg60Core(input).toUpperCase();
}

/**
 * Tuong duong reg61() trong VBA cu: MD5 chuan cua chuoi, 32 ky tu chu HOA.
 * Vi du: =REG61(A1)
 * @param {string} input Chuoi bat ky can hash
 * @return {string} MD5 dang hex 32 ky tu chu HOA
 * @customfunction
 */
function REG61(input) {
  if (input === undefined || input === null || input === '') return '';
  return MD5_HEX(input).toUpperCase();
}

/**
 * Tuong duong reg70() trong VBA cu: MD5 chuan, sau do lay 8 ky tu
 * tai vi tri 1,5,9,13,17,21,25,29 (dem tu 1) cua chuoi MD5 32 ky tu HOA.
 * Vi du: =REG70(A1)
 * @param {string} input Chuoi bat ky can hash
 * @return {string} Chuoi 8 ky tu rut gon tu MD5
 * @customfunction
 */
function REG70(input) {
  if (input === undefined || input === null || input === '') return '';
  var full = MD5_HEX(input).toUpperCase();
  // vi tri 1-indexed: 1,5,9,13,17,21,25,29 -> 0-indexed: 0,4,8,12,16,20,24,28
  var idxs = [0, 4, 8, 12, 16, 20, 24, 28];
  var out = '';
  for (var i = 0; i < idxs.length; i++) {
    out += full.charAt(idxs[i]);
  }
  return out;
}

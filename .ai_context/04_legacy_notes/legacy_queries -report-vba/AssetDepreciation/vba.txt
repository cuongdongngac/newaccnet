Private Function Refreshing()
Dim SQL
Dim MyDB As DAO.Database
Set MyDB = CurrentDb()
DoCmd.SetWarnings False
m = SaveRecord()
DoCmd.RunSQL "DELETE DISTINCTROW * FROM [Depreciation Table];"

SQL = "INSERT INTO [Depreciation Table] ( [Asset ID], SrCode,Rest,RestQty, Amount, Flag ) "
SQL = SQL & "SELECT DISTINCTROW [Union Asset Line Query].[Asset ID], [Union Asset Line Query].[SrCode],"
SQL = SQL & "Sum([Union Asset Line Query]![Quantity]*"
SQL = SQL & "([Union Asset Line Query]![Price]-[Union Asset Line Query]![RemainValue])*"
SQL = SQL & "[Union Asset Line Query]![DBCR]) AS Rest, "

SQL = SQL & "Sum(-[Union Asset Line Query]![Quantity]*"
SQL = SQL & "[Union Asset Line Query]![Flag]*"
SQL = SQL & "[Union Asset Line Query]![DBCR]) AS RestQty, "

SQL = SQL & "First(GetDepreciation(GetBeginDate(),GetEndDate(),[Asset List]![Date],[Asset List]![TimeUsing],[AssetTracking]![DepRate],[AssetTracking]![DepPrice])) "
SQL = SQL & "AS Amount, First(-1) AS Flag "
SQL = SQL & "FROM ([Union Asset Line Query] INNER JOIN [Asset List] "
SQL = SQL & "ON [Union Asset Line Query].[Asset ID] = [Asset List].[Asset ID])INNER JOIN AssetTracking ON [Asset List].[Asset ID] = AssetTracking.[Asset ID] "
SQL = SQL & "WHERE [Union Asset Line Query].[Bill Date] <=GetEndDate() "
SQL = SQL & "GROUP BY [Union Asset Line Query].[Asset ID],[Union Asset Line Query].[SrCode] "
SQL = SQL & "ORDER BY [Union Asset Line Query].[Asset ID];"

DoCmd.RunSQL SQL
Me![Depreciation SubForm].Form.Requery
End Function

Public Function AddingExAccountID(ExAccountID As String, ExObjectID As String, Amount As Double)

Dim MyDB As DAO.Database
Dim ExID As Long
Dim TransRecord As DAO.Recordset
Dim ExRecord As DAO.Recordset
Set MyDB = CurrentDb()
 Set TransRecord = MyDB.OpenRecordset("Transaction List")
 Set ExRecord = MyDB.OpenRecordset("Expense Object Input Line")

TransRecord("Account ID") = ExAccountID
TransRecord("DBCR") = 1
TransRecord("Bill ID") = gBillID
TransRecord("Amount") = Amount

'them code thinhpv
 TransRecord("CreateDate") = Now

 'end code thinhpv

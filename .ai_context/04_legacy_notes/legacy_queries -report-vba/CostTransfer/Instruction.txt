SELECT DISTINCTROW [Union Expense Object Line Query].[Account ID], [Union Expense Object Line Query].[ExObject ID], [Union Expense Object Line Query].[Expense Atribute ID], Sum([Union Expense Object Line Query]![DBCR]*[Union Expense Object Line Query]![Amount]) AS Rest, -1 AS Flag
FROM [Union Expense Object Line Query]
WHERE ((([Union Expense Object Line Query].[Bill Date])<=GetEnddate()))
GROUP BY [Union Expense Object Line Query].[Account ID], [Union Expense Object Line Query].[ExObject ID], [Union Expense Object Line Query].[Expense Atribute ID]
HAVING (((Sum([Union Expense Object Line Query]![DBCR]*[Union Expense Object Line Query]![Amount]))<>0));

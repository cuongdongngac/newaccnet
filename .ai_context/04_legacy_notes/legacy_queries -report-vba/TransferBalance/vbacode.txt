Public Sub AssetBalanceInput(DesDatabaseFile)
Dim i As Integer
Dim Recno As Variant
Dim SQL As String
Dim desdb As DAO.Database
Dim MyDB As DAO.Database
Dim MyRecordset As DAO.Recordset
Dim TransactionRecordset As DAO.Recordset
Dim result As Variant

Set MyDB = CurrentDb()
Set desdb = DBEngine.Workspaces(0).OpenDatabase(DesDatabaseFile)
Set TransactionRecordset = desdb.OpenRecordset("Transaction List")
TransactionRecordset.MoveLast
Recno = TransactionRecordset.RecordCount

Set MyRecordset = MyDB.OpenRecordset("Asset Balance Account Query")
If MyRecordset.RecordCount = 0 Then
 TransactionRecordset.Close
 MyRecordset.Close
 MyDB.Close
 desdb.Close
 Exit Sub
End If
SysCmd acSysCmdInitMeter, " Transfering Aset Balance...", MyRecordset.RecordCount
MyRecordset.MoveFirst
'Update balance for accounts with debit amount

i = 1
Do While Not MyRecordset.EOF

 If MyRecordset("DBAmount") <> 0 Then
  TransactionRecordset.AddNew
  TransactionRecordset("Account ID") = MyRecordset("Account ID")
  TransactionRecordset("Amount") = MyRecordset("DBAmount")
  TransactionRecordset("DBCR") = 1
  TransactionRecordset.Update
 ' Recno = Recno + 1
  TransactionRecordset.MoveLast
  Recno = TransactionRecordset("ID")
  
  SQL = "INSERT INTO [Asset Input Line] ( ID, [Asset ID], [Quantity],Price,  ReasonID, SrCode)  IN  " & "'" & DesDatabaseFile & "' "
  SQL = SQL & " SELECT DISTINCTROW  " & Recno & " AS ID, [Asset Balance Line Query].[Asset ID] as [Asset ID], "
  SQL = SQL & " [Asset Balance Line Query].[Quantity] as [Quantity],"
  SQL = SQL & " [Asset Balance Line Query].[Amount]/[Asset Balance Line Query].[Quantity]  as [Price], "
  SQL = SQL & "[Asset Balance Line Query].[ReasonID] as ReasonID, "
  SQL = SQL & "[Asset Balance Line Query].[SrCode] as SrCode "
  SQL = SQL & " FROM [Asset Balance Line Query] "
  SQL = SQL & " WHERE [Asset Balance Line Query].[Quantity] >0  And "
  SQL = SQL & " [Asset Balance Line Query].[Account ID]=" & "'" & MyRecordset("Account ID") & "';"
  MyDB.Execute (SQL)
  
  End If

 If MyRecordset("CRAmount") <> 0 Then
  TransactionRecordset.AddNew
  TransactionRecordset("Account ID") = MyRecordset("Account ID")
  TransactionRecordset("Amount") = MyRecordset("CRAmount")
  TransactionRecordset("DBCR") = -1
  TransactionRecordset.Update
  'Recno = Recno + 1
  TransactionRecordset.MoveLast
  Recno = TransactionRecordset("ID")
  
  SQL = "INSERT INTO [Asset Input Line] ( ID, [Asset ID], [Quantity],Price,  ReasonID, SrCode)  IN  " & "'" & DesDatabaseFile & "' "
  SQL = SQL & " SELECT DISTINCTROW  " & Recno & " AS ID, [Asset Balance Line Query].[Asset ID] as [Asset ID], "
  SQL = SQL & " -[Asset Balance Line Query].[Quantity] as [Quantity],"
  SQL = SQL & " [Asset Balance Line Query].[Amount]/[Asset Balance Line Query].[Quantity]  as [Price], "
  SQL = SQL & "[Asset Balance Line Query].[ReasonID] as ReasonID, "
  SQL = SQL & "[Asset Balance Line Query].[SrCode] as SrCode "
  SQL = SQL & " FROM [Asset Balance Line Query] "
  SQL = SQL & " WHERE [Asset Balance Line Query].[Quantity] < 0  And "
  SQL = SQL & " [Asset Balance Line Query].[Account ID]=" & "'" & MyRecordset("Account ID") & "';"
  MyDB.Execute (SQL)
 End If
  MyRecordset.MoveNext
  i = i + 1
SysCmd acSysCmdUpdateMeter, i
Loop

TransactionRecordset.Close
MyRecordset.Close
MyDB.Close
desdb.Close

SysCmd acSysCmdRemoveMeter

End Sub
